# Exercise the production on-cast event sequence without starting Skyrim.
set(ABW_ACTIVATE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../src/Activate.cpp" CACHE FILEPATH "Activation source under test")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${ABW_ACTIVATE_SOURCE}")
file(READ "${ABW_ACTIVATE_SOURCE}" activateSource)
set(activateFunctions "")
foreach(pair IN ITEMS "constexpr float kHeldBoundSeconds|struct BoundCopies" "struct OnCastCoalesce|void FinishOnCastSpawn" "bool HasFreshDualBoundEffect|RE::SpellItem* AssignableBoundOnPlayer")
    string(REPLACE "|" ";" markers "${pair}")
    list(GET markers 0 beginMarker)
    list(GET markers 1 endMarker)
    string(FIND "${activateSource}" "${beginMarker}" beginOffset)
    string(FIND "${activateSource}" "${endMarker}" endOffset)
    if(beginOffset LESS 0 OR endOffset LESS beginOffset)
        message(FATAL_ERROR "Activation test extraction markers changed")
    endif()
    math(EXPR sectionLength "${endOffset} - ${beginOffset}")
    string(SUBSTRING "${activateSource}" ${beginOffset} ${sectionLength} section)
    string(APPEND activateFunctions "${section}")
endforeach()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ActivateFunctions.inc" "${activateFunctions}")
add_executable(ActivateTests ActivateTests.cpp)
target_compile_features(ActivateTests PRIVATE cxx_std_23)
target_include_directories(ActivateTests PRIVATE . "${CMAKE_CURRENT_BINARY_DIR}")
add_test(NAME ActivateTests COMMAND ActivateTests)
